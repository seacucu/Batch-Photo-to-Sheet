function onOpen() {

  const ui = SpreadsheetApp.getUi();

  ui.createMenu("Chloe的工具")
    .addItem("批次上傳照片", "showFolderPrompt")
    .addToUi();
}


function showFolderPrompt() {

  const ui = SpreadsheetApp.getUi();

  const response = ui.prompt(
    "批次上傳",
    "請輸入 Folder ID",
    ui.ButtonSet.OK_CANCEL
  );

  if (response.getSelectedButton() != ui.Button.OK) {
    return;
  }

  var folderId = response.getResponseText();

  importImagesFromFolder(folderId);
}


function importImagesFromFolder(folderId) {

  // var folderId = 'FolderID'; // replace with FolderID
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var folder = DriveApp.getFolderById(folderId);
  var files = folder.getFiles();
  
  var fileList = [];
  
  // Fetch file information
  while (files.hasNext()) {
    var file = files.next();
    var mimeType = file.getMimeType();
    if (mimeType.indexOf("image/") !== -1) {
      fileList.push({
        name: file.getName(),
        id: file.getId()
      });
    }
  }
  
  // Sort by file name
  fileList.sort(function(a, b) {
    return a.name.localeCompare(b.name, undefined, {numeric: true, sensitivity: 'base'});
  });
  
  // Initialize sheet and write headers
  sheet.clear();
  sheet.appendRow(["Image", "File Name", "tempURL"]); 
  
  var dataToRows = [];
  
  for (var i = 0; i < fileList.length; i++) {
    var fileId = fileList[i].id;
    var fileName = fileList[i].name;

    // Use high-resolution thumbnail URL (1000px)
    var imageUrl = 'https://drive.google.com/thumbnail?id=' + fileId;
    var formula = '=IMAGE("' + imageUrl + '")';
    
    // Temporarily place formula in the third column (Column C)
    dataToRows.push(["", fileName, formula]);
  }
  
  if (dataToRows.length > 0) {

    var range = sheet.getRange(2, 1, dataToRows.length, 3);
    range.setValues(dataToRows);
    
    // Force spreadsheet to complete formula calculations
    SpreadsheetApp.flush();
    
    // Convert formulas in Column C to static images in Column A (Paste Values)
    var formulaRange = sheet.getRange(2, 3, dataToRows.length, 1);
    var targetStaticRange = sheet.getRange(2, 1, dataToRows.length, 1);
    
    // Copy images generated by formulas and paste as values to Column A
    formulaRange.copyTo(targetStaticRange, SpreadsheetApp.CopyPasteType.PASTE_VALUES, false);
    
    // Delete the temporary helper Column C
    sheet.deleteColumn(3);
  }
}
